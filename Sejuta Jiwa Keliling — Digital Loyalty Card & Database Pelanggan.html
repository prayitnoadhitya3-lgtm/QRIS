                Otomatisasi 1-Klik di Google Sheets
                
Skrip Pembuat Google Form Otomatis

                
Salin kode ini ke menu Extensions > Apps Script di Google Sheets-mu, lalu klik Run pada fungsi setupOtomatisSejutaJiwa. Google Form akan langsung terbuat dan terhubung otomatis ke tab database!

              

              
            


            
function setupOtomatisSejutaJiwa() {
  const ss = SpreadsheetApp.openById('167cBPm4BW1OnUHXEexm1_hlXxp4Gtl1_PFsZaN8bRes');
  const form = FormApp.create('Pendaftaran Member Loyalty - Sejuta Jiwa Keliling');
  form.setDescription('Daftar Member Gratis Sejuta Jiwa Keliling! Kumpulkan 9 Stempel Gratis 1 Cup Kopi + Gratis 1 Cup Kopi di Hari Ulang Tahunmu!');
  form.setConfirmationMessage('Terima kasih sudah bergabung di Member Sejuta Jiwa Keliling! Stempel pertamamu sudah tercatat.');

  form.addTextItem().setTitle('Nama Lengkap').setRequired(true);
  form.addTextItem().setTitle('Nomor WhatsApp (Aktif)').setRequired(true);
  form.addDateItem().setTitle('Tanggal Lahir (Untuk Promo Gratis Kopi Ultah)').setRequired(true);
  form.addTextItem().setTitle('Lokasi / Titik Mangkal Langganan').setRequired(false);
  form.addMultipleChoiceItem()
      .setTitle('Menu Kopi Favorit')
      .setChoiceValues(['Kopi Susu Gula Aren', 'Cold Brew Black 250ml', 'Matcha Latte Oatmilk', 'Americano Double Shot', 'Kopi Susu Pandan', 'Caramel Macchiato'])
      .showOtherOption(true)
      .setRequired(true);
  form.addTextItem().setTitle('Catatan Preferensi (Takaran Gula / Es)').setRequired(false);

  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());
  Logger.log('Link Google Form Pelanggan: ' + form.getPublishedUrl());
}
